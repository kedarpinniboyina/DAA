#include <stdio.h>
#include <stdlib.h>

struct Node{
    int data;
    struct Node *next;
};

int main(){
    
    struct Node * head= NULL;
    struct Node *newNode;
    struct Node *temp;
    
    int n,i;
    printf("Enter the No of Nodes");
    scanf("%d", &n);
    
    for( i=1;i<=n;i++){
        newNode=(struct Node *)malloc(sizeof(struct Node));
        
        printf("Enter the Data");
    
        scanf("%d", &newNode->data);
        
        newNode->next=NULL;
    
        if(head==NULL){
            head =newNode;
        }
        else{
           newNode->next=head;
           head=newNode;
        }
    }
    
    
    printf("Linked List");
    temp=head;
    while(temp!=NULL){
        printf("%d ->", temp->data);
        temp=temp->next;
    }
    printf("NULL");
    return 0;
}
