#include <stdio.h>

#define INF 999

int main()
{
    int n, i, j, source;
    int cost[10][10], dist[10], visited[10];
    int count, u, min;

    printf("Enter number of vertices: ");
    scanf("%d", &n);

    printf("Enter cost adjacency matrix:\n");

    for(i = 0; i < n; i++)
    {
        for(j = 0; j < n; j++)
        {
            scanf("%d", &cost[i][j]);

            if(cost[i][j] == 0 && i != j)
                cost[i][j] = INF;
        }
    }

    printf("Enter source vertex: ");
    scanf("%d", &source);

    // Initialize
    for(i = 0; i < n; i++)
    {
        dist[i] = cost[source][i];
        visited[i] = 0;
    }

    dist[source] = 0;

    // Dijkstra Algorithm
    for(count = 0; count < n; count++)
    {
        min = INF;
        u = -1;

        // Find minimum distance vertex
        for(i = 0; i < n; i++)
        {
            if(visited[i] == 0 && dist[i] < min)
            {
                min = dist[i];
                u = i;
            }
        }

        if(u == -1)
            break;

        visited[u] = 1;

        // Update distances
        for(i = 0; i < n; i++)
        {
            if(visited[i] == 0 &&
               dist[u] + cost[u][i] < dist[i])
            {
                dist[i] = dist[u] + cost[u][i];
            }
        }
    }

    printf("\nShortest distances from vertex %d:\n", source);

    for(i = 0; i < n; i++)
    {
        printf("Vertex %d = %d\n", i, dist[i]);
    }

    return 0;
}
